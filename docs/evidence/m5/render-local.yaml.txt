apiVersion: apps/v1
kind: Deployment
metadata:
  name: urumi-m5-harness
spec:
  replicas: 1
  strategy: {type: Recreate}
  selector:
    matchLabels: {app.kubernetes.io/instance: urumi, app.kubernetes.io/component: backend}
  template:
    metadata:
      labels: {app.kubernetes.io/name: urumi, app.kubernetes.io/instance: urumi, app.kubernetes.io/component: backend}
    spec:
      serviceAccountName: urumi-backend
      securityContext: {runAsNonRoot: true, runAsUser: 10001, runAsGroup: 10001, seccompProfile: {type: RuntimeDefault}}
      containers:
        - name: backend
          image: "urumi/m5-harness:m5"
          imagePullPolicy: IfNotPresent
          command: [python, -m, scripts.m5.harness]
          securityContext: {readOnlyRootFilesystem: true, allowPrivilegeEscalation: false, capabilities: {drop: [ALL]}}
          ports: [{name: http, containerPort: 8000}]
          env:
            - {name: PGHOST, value: "urumi-platform-db"}
            - {name: PGPORT, value: "5432"}
            - {name: PGUSER, value: urumi}
            - {name: PGDATABASE, value: urumi}
            - name: PGPASSWORD
              valueFrom:
                secretKeyRef: {name: urumi-platform-db, key: password}
            - {name: M5_ORIGIN, value: "https://gateway.example.test"}
            - {name: M5_LOCAL_HTTP, value: "false"}
            - {name: M5_SECRET_NAMESPACE, value: "urumi-upstream-secrets"}
            - {name: M5_INTERNAL_URLS, value: "[\"http://urumi-woocommerce-mcp.urumi.svc.cluster.local:8000/mcp\",\"http://urumi-inventory-mcp.urumi.svc.cluster.local:8000/mcp\"]"}
            - {name: M5_LISTEN_HOST, value: "0.0.0.0"}
          startupProbe:
            httpGet: {path: /health/live, port: http}
            periodSeconds: 2
            timeoutSeconds: 2
            failureThreshold: 60
          readinessProbe:
            httpGet: {path: /health/ready, port: http}
            periodSeconds: 5
            timeoutSeconds: 5
          livenessProbe:
            httpGet: {path: /health/live, port: http}
            periodSeconds: 10
            timeoutSeconds: 2
          resources:
            limits:
              cpu: 1000m
              memory: 512Mi
            requests:
              cpu: 100m
              memory: 256Mi

# Source: urumi/templates/platform-db/database.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: urumi-platform-db
spec:
  replicas: 1
  strategy: {type: Recreate}
  selector:
    matchLabels: {app.kubernetes.io/instance: urumi, app.kubernetes.io/component: platform-db}
  template:
    metadata:
      labels: {app.kubernetes.io/instance: urumi, app.kubernetes.io/component: platform-db}
    spec:
      automountServiceAccountToken: false
      containers:
        - name: postgres
          image: "postgres@sha256:3645570cccdfa447589da9f57dd740faa29b30938e861289a5574b6ca6b03826"
          env:
            - {name: POSTGRES_USER, value: urumi}
            - {name: POSTGRES_DB, value: urumi}
            - name: POSTGRES_PASSWORD
              valueFrom:
                secretKeyRef: {name: urumi-platform-db, key: password}
            - {name: PGDATA, value: /var/lib/postgresql/data/pgdata}
          ports: [{name: postgres, containerPort: 5432}]
          startupProbe:
            exec: {command: [pg_isready, -U, urumi, -d, urumi]}
            periodSeconds: 2
            failureThreshold: 60
          readinessProbe:
            exec: {command: [pg_isready, -U, urumi, -d, urumi]}
            periodSeconds: 3
          livenessProbe:
            exec: {command: [pg_isready, -U, urumi, -d, urumi]}
            periodSeconds: 10
          resources: {"limits":{"cpu":"500m","memory":"512Mi"},"requests":{"cpu":"100m","memory":"128Mi"}}
          volumeMounts: [{name: data, mountPath: /var/lib/postgresql/data}]
      volumes:
        - name: data
          persistentVolumeClaim: {claimName: urumi-platform-data}
