# Source: urumi/templates/foundation/configmap.yaml
apiVersion: v1
kind: ConfigMap
metadata:
  name: urumi-foundation-config
  labels:
    app.kubernetes.io/name: urumi
    app.kubernetes.io/instance: urumi
    app.kubernetes.io/component: foundation
    app.kubernetes.io/managed-by: Helm
    helm.sh/chart: "urumi-0.1.0"
data:
  index.html: |
    <!doctype html>
    <html lang="en">
      <head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>Urumi M1</title></head>
      <body><h1>Urumi M1 — Kubernetes is serving this page.</h1><p>This temporary page checks routing, saved storage, and connection rules.</p></body>
    </html>
  server.py: |
    from functools import partial
    from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
    from pathlib import Path
    import shutil

    data = Path("/data")
    index = data / "index.html"
    if not index.exists():
        shutil.copyfile("/config/index.html", index)
    handler = partial(SimpleHTTPRequestHandler, directory=str(data))
    ThreadingHTTPServer(("0.0.0.0", 8080), handler).serve_forever()
