# Source: urumi/templates/store/wordpress.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: urumi-wordpress
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app.kubernetes.io/instance: urumi
      app.kubernetes.io/component: wordpress
  template:
    metadata:
      labels:
        app.kubernetes.io/instance: urumi
        app.kubernetes.io/component: wordpress
    spec:
      automountServiceAccountToken: false
      securityContext:
        fsGroup: 33
      containers:
        - name: wordpress
          image: "private-registry.example.com/urumi-store:replace-with-immutable-reference"
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 80
          envFrom:
            - configMapRef:
                name: urumi-store
          env:
            - {name: WORDPRESS_DB_PASSWORD_FILE, value: /run/urumi/database/database-password}
            - name: WORDPRESS_AUTH_KEY_FILE
              value: /run/urumi/salts/AUTH_KEY
            - name: WORDPRESS_SECURE_AUTH_KEY_FILE
              value: /run/urumi/salts/SECURE_AUTH_KEY
            - name: WORDPRESS_LOGGED_IN_KEY_FILE
              value: /run/urumi/salts/LOGGED_IN_KEY
            - name: WORDPRESS_NONCE_KEY_FILE
              value: /run/urumi/salts/NONCE_KEY
            - name: WORDPRESS_AUTH_SALT_FILE
              value: /run/urumi/salts/AUTH_SALT
            - name: WORDPRESS_SECURE_AUTH_SALT_FILE
              value: /run/urumi/salts/SECURE_AUTH_SALT
            - name: WORDPRESS_LOGGED_IN_SALT_FILE
              value: /run/urumi/salts/LOGGED_IN_SALT
            - name: WORDPRESS_NONCE_SALT_FILE
              value: /run/urumi/salts/NONCE_SALT
          resources:
            limits:
              cpu: 1000m
              memory: 768Mi
            requests:
              cpu: 100m
              memory: 256Mi
          startupProbe:
            httpGet: {path: /urumi-live.php, port: 80}
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 36
          livenessProbe:
            httpGet: {path: /urumi-live.php, port: 80}
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 6
          readinessProbe:
            httpGet: {path: /urumi-ready.php, port: 80}
            periodSeconds: 10
            timeoutSeconds: 3
          volumeMounts:
            - {name: content, mountPath: /var/www/html}
            - {name: database, mountPath: /run/urumi/database, readOnly: true}
            - {name: salts, mountPath: /run/urumi/salts, readOnly: true}
      volumes:
        - name: content
          persistentVolumeClaim:
            claimName: urumi-wordpress-content
        - name: database
          secret:
            secretName: urumi-store-db
            items:
              - {key: database-password, path: database-password}
        - name: salts
          secret:
            secretName: urumi-store-salts

      securityContext: {runAsNonRoot: true, runAsUser: 10001, runAsGroup: 10001, seccompProfile: {type: RuntimeDefault}}

          securityContext: {readOnlyRootFilesystem: true, allowPrivilegeEscalation: false, capabilities: {drop: [ALL]}}
          ports: [{name: mcp, containerPort: 8000}]
          env:
            - {name: WOO_CANONICAL_ORIGIN, value: "https://shop.example.com"}
            - {name: WOO_TRANSPORT_ORIGIN, value: "https://shop.example.com"}
            - {name: WOO_CURRENCY, value: "INR"}
            - {name: MCP_ALLOWED_HOSTS, value: "urumi-woocommerce-mcp,urumi-woocommerce-mcp.urumi.svc.cluster.local"}
            - name: WOO_CONSUMER_KEY
              valueFrom:
                secretKeyRef: {name: urumi-woocommerce-rest, key: consumer-key}

                secretKeyRef: {name: urumi-woocommerce-rest, key: consumer-secret}

  name: urumi-mariadb

  serviceName: urumi-mariadb
  replicas: 1
  selector:
    matchLabels:
      app.kubernetes.io/instance: urumi
      app.kubernetes.io/component: mariadb
  template:
    metadata:
      labels:
        app.kubernetes.io/instance: urumi
        app.kubernetes.io/component: mariadb
    spec:
      automountServiceAccountToken: false
      containers:
        - name: mariadb
          image: "mariadb:11.4.13-noble"
          ports:
            - containerPort: 3306
          env:
            - {name: MARIADB_DATABASE, value: woocommerce}
            - {name: MARIADB_USER, value: woocommerce}
            - {name: MARIADB_PASSWORD_FILE, value: /run/urumi/database/database-password}
            - {name: MARIADB_ROOT_PASSWORD_FILE, value: /run/urumi/database/root-password}
          resources:
            limits:
              cpu: 1000m
              memory: 768Mi
            requests:
              cpu: 100m
              memory: 256Mi
          startupProbe:
            exec:
              command: [healthcheck.sh, --connect, --innodb_initialized]
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 36
          readinessProbe:
            exec:
              command: [healthcheck.sh, --connect, --innodb_initialized]
            periodSeconds: 5
            timeoutSeconds: 3
          livenessProbe:
            exec:
              command: [healthcheck.sh, --connect]
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 6
          volumeMounts:
            - {name: data, mountPath: /var/lib/mysql}
            - {name: database, mountPath: /run/urumi/database, readOnly: true}
      volumes:
        - name: data
          persistentVolumeClaim:
            claimName: urumi-mariadb-data
        - name: database
